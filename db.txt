CREATE TABLE Officina (
    codice CHAR(5) PRIMARY KEY,
    denominazione VARCHAR(100) NOT NULL,
    indirizzo VARCHAR(200) NOT NULL,
    telefono VARCHAR(20),
    centrale BOOLEAN
);

CREATE TABLE Pezzo (
    codice_pezzo VARCHAR(20) NOT NULL,
    descrizione TEXT NOT NULL,
    costo_unitario DECIMAL(10,2) NOT NULL,
    quantita INT NOT NULL,
    officina_codice CHAR(5) NOT NULL,
    PRIMARY KEY (codice_pezzo, officina_codice),
    FOREIGN KEY (officina_codice) REFERENCES Officina(codice)
);

CREATE TABLE Articolo (
    codice_articolo VARCHAR(20) NOT NULL,
    descrizione TEXT NOT NULL,
    costo_unitario DECIMAL(10,2) NOT NULL,
    quantita INT NOT NULL,
    officina_codice CHAR(5) NOT NULL,
    PRIMARY KEY (codice_articolo, officina_codice),
    FOREIGN KEY (officina_codice) REFERENCES Officina(codice)
);

CREATE TABLE Servizio (
    codice_servizio VARCHAR(20) NOT NULL,
    descrizione TEXT NOT NULL,
    costo_orario DECIMAL(10,2) NOT NULL,
    officina_codice CHAR(5) NOT NULL,
    PRIMARY KEY (codice_servizio, officina_codice),
    FOREIGN KEY (officina_codice) REFERENCES Officina(codice)
);

CREATE TABLE Cliente (
    id_cliente INT PRIMARY KEY AUTO_INCREMENT,
    cognome VARCHAR(50) NOT NULL,
    nome VARCHAR(50) NOT NULL,
    telefono VARCHAR(20),
    email VARCHAR(100),
    indirizzo VARCHAR(200)
);


CREATE TABLE Veicolo (
    targa VARCHAR(10) PRIMARY KEY,
    numero_telaio VARCHAR(30) NOT NULL,
    anno_costruzione INT,
    marca VARCHAR(50) NOT NULL,
    modello VARCHAR(50) NOT NULL
);

CREATE TABLE TipoIntervento (
    id_tipo_intervento INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(50) NOT NULL,
    descrizione TEXT
);

CREATE TABLE Intervento (
    id_intervento INT PRIMARY KEY AUTO_INCREMENT,
    data_prenotazione TIMESTAMP NOT NULL,
    data_effettiva DATE,
    stato VARCHAR(20) NOT NULL,
    note TEXT,
    ore_lavorate DECIMAL(5,2),
    costo_totale DECIMAL(10,2),
    officina_codice CHAR(5) NOT NULL,
    id_cliente INT NOT NULL,
    targa_veicolo VARCHAR(10) NOT NULL,
    id_tipo_intervento INT NOT NULL,
    FOREIGN KEY (officina_codice) REFERENCES Officina(codice),
    FOREIGN KEY (id_cliente) REFERENCES Cliente(id_cliente),
    FOREIGN KEY (targa_veicolo) REFERENCES Veicolo(targa),
    FOREIGN KEY (id_tipo_intervento) REFERENCES TipoIntervento(id_tipo_intervento)
);
CREATE TABLE users (
    id       INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    email    VARCHAR(100) UNIQUE NOT NULL,
    ruolo    VARCHAR(20) NOT NULL DEFAULT 'utente',
    verified BOOLEAN NOT NULL DEFAULT FALSE,
    verification_token VARCHAR(255)
);
